        async function sendManagerPrompt(prompt) {
            const chatHistory = document.getElementById('manager-chat-history');
            
            // 1. Add User Message to UI
            const userMsgHTML = `
                <div class="flex items-start gap-3 justify-end">
                    <div class="bg-brand-600 p-3 rounded-lg shadow-sm text-sm text-white max-w-[80%]">
                        ${prompt}
                    </div>
                </div>
            `;
            chatHistory.insertAdjacentHTML('beforeend', userMsgHTML);
            chatHistory.scrollTop = chatHistory.scrollHeight;

            // 2. Add Loading Indicator
            const loadingId = 'loading-' + Date.now();
            const loadingHTML = `
                <div id="${loadingId}" class="flex items-start gap-3">
                    <div class="w-8 h-8 bg-brand-500 rounded-full flex items-center justify-center text-white flex-shrink-0">
                        <i class="fa-solid fa-robot text-xs"></i>
                    </div>
                    <div class="bg-white p-3 rounded-lg border border-slate-200 shadow-sm text-sm text-slate-500 flex items-center gap-2">
                        <i class="fa-solid fa-circle-notch fa-spin text-brand-500"></i> Analizando datos del equipo...
                    </div>
                </div>
            `;
            chatHistory.insertAdjacentHTML('beforeend', loadingHTML);
            chatHistory.scrollTop = chatHistory.scrollHeight;

            // 3. Call Gemini API
            const systemInstruction = "Eres un asistente de IA para managers llamado SmartOps. Tu objetivo es proporcionar información clara, concisa y procesable sobre la carga de trabajo del equipo basándote SÓLO en el contexto proporcionado. Usa un tono profesional, empático y orientado a prevenir el burnout. Formatea tu respuesta con listas o negritas si es útil, pero mantenla breve (máximo 2-3 párrafos). Devuelve la respuesta en HTML seguro (sin etiquetas de bloque completas, solo formato de texto dentro de divs/p/ul).";
            
            const fullQuery = `
                Contexto:
                ${managerChatContext}
                
                Pregunta del Manager:
                ${prompt}
            `;

            const apiKey = ""; // Handled by environment
            const apiUrl = `https://generativelanguage.googleapis.com/v1beta/models/gemini-3-flash-preview:generateContent?key=${apiKey}`;

            const payload = {
                contents: [{ parts: [{ text: fullQuery }] }],
                systemInstruction: { parts: [{ text: systemInstruction }] }
            };

            try {
                const response = await fetch(apiUrl, {
                    method: 'POST',
                    headers: { 'Content-Type': 'application/json' },
                    body: JSON.stringify(payload)
                });

                if (!response.ok) throw new Error(`HTTP error! status: ${response.status}`);
                const result = await response.json();
                const candidate = result.candidates?.[0];

                // Remove loading indicator
                document.getElementById(loadingId).remove();

                if (candidate && candidate.content?.parts?.[0]?.text) {
                    let aiResponse = candidate.content.parts[0].text;
                    // Clean up potential markdown formatting that shouldn't be there
                     aiResponse = aiResponse.replace(/^```html\n?/, '').replace(/\n?```$/, '');
                     
                     // Helper function to render bold text if the model returned markdown bold instead of html
                     aiResponse = aiResponse.replace(/\*\*(.*?)\*\*/g, '<strong>$1</strong>');
                     aiResponse = aiResponse.replace(/\*(.*?)\*/g, '<em>$1</em>');
                     
                     // Basic line break handling if model didn't return HTML paragraphs
                     if(!aiResponse.includes('<p>') && !aiResponse.includes('<br>')) {
                         aiResponse = aiResponse.replace(/\n/g, '<br>');
                     }

                    // 4. Add AI Response to UI
                    const aiMsgHTML = `
                        <div class="flex items-start gap-3">
                            <div class="w-8 h-8 bg-brand-500 rounded-full flex items-center justify-center text-white flex-shrink-0">
                                <i class="fa-solid fa-robot text-xs"></i>
                            </div>
                            <div class="bg-white p-3 rounded-lg border border-slate-200 shadow-sm text-sm text-slate-700 max-w-[80%] space-y-2">
                                ${aiResponse}
                            </div>
                        </div>
                    `;
                    chatHistory.insertAdjacentHTML('beforeend', aiMsgHTML);
                }
            } catch (error) {
                console.error("Error calling Gemini API:", error);
                document.getElementById(loadingId).remove();
                
                const errorMsgHTML = `
                    <div class="flex items-start gap-3">
                        <div class="w-8 h-8 bg-brand-500 rounded-full flex items-center justify-center text-white flex-shrink-0">
                            <i class="fa-solid fa-robot text-xs"></i>
                        </div>
                        <div class="bg-rose-50 p-3 rounded-lg border border-rose-200 shadow-sm text-sm text-rose-700">
                            <i class="fa-solid fa-triangle-exclamation mr-1"></i> Lo siento, hubo un problema al procesar tu solicitud. Inténtalo de nuevo.
                        </div>
                    </div>
                `;
                chatHistory.insertAdjacentHTML('beforeend', errorMsgHTML);
            }
            
            chatHistory.scrollTop = chatHistory.scrollHeight;
        }

        async function callGemini(userInstruction) {
            const loadingIndicator = document.getElementById('ai-loading');
            const panel = document.getElementById('ai-assistant-panel');
            
            // Extract the current report content
            const reportContainer = document.querySelector('.p-8.space-y-6.text-sm.text-slate-700');
            const currentReportContent = reportContainer.innerText;

            loadingIndicator.classList.remove('hidden');

            const systemPrompt = "Actúa como un asistente virtual corporativo llamado SmartOps. Tu tarea es reescribir y mejorar el reporte de estado de proyecto que se te proporciona, siguiendo estrictamente la instrucción del usuario. Devuelve ÚNICAMENTE el reporte reescrito en formato HTML, utilizando las mismas clases de Tailwind CSS y estructura que el original (secciones con bordes, iconos de FontAwesome, listas ul/li). No incluyas explicaciones previas ni bloques de markdown (```html). Mantén el tono profesional.";
            
            const fullQuery = `
                Instrucción: ${userInstruction}
                
                Reporte Original:
                ${currentReportContent}
            `;

            const apiKey = ""; // API key is handled automatically by the environment
            const apiUrl = `https://generativelanguage.googleapis.com/v1beta/models/gemini-3-flash-preview:generateContent?key=${apiKey}`;

            const payload = {
                contents: [{ parts: [{ text: fullQuery }] }],
                systemInstruction: { parts: [{ text: systemPrompt }] }
            };

            try {
                const response = await fetch(apiUrl, {
                    method: 'POST',
                    headers: { 'Content-Type': 'application/json' },
                    body: JSON.stringify(payload)
                });

                if (!response.ok) {
                    throw new Error(`HTTP error! status: ${response.status}`);
                }

                const result = await response.json();
                const candidate = result.candidates?.[0];

                if (candidate && candidate.content?.parts?.[0]?.text) {
                    let newHTML = candidate.content.parts[0].text;
                    
                    // Basic cleanup in case the model wraps it in markdown anyway
                    newHTML = newHTML.replace(/^```html\n?/, '').replace(/\n?```$/, '');

                    // Replace the content
                    reportContainer.innerHTML = newHTML;
                    
                    // Hide panel and clear loading
                    toggleAIAssistant();
                    document.getElementById('ai-custom-prompt').value = '';
                } else {
                    console.error("Unexpected API response format:", result);
                    alert("Error: No se recibió una respuesta válida de la IA.");
                }

            } catch (error) {
                console.error("Error calling Gemini API:", error);
                alert("Hubo un problema al conectar con el asistente de IA. Inténtalo de nuevo más tarde.");
            } finally {
                loadingIndicator.classList.add('hidden');
            }
        }

        function runGlobalAutomation() {
            const btnRun = document.getElementById('btn-run-automation');

            const progressBar = document.getElementById('progress-bar');
            const progressText = document.getElementById('progress-text');
            const progressPercent = document.getElementById('progress-percent');

            // 1. Ocultar botón, mostrar progreso

            progressContainer.classList.remove('hidden');
            resultsPanel.classList.add('hidden');
            resultsPanel.classList.remove('opacity-100', 'translate-y-0');
            resultsPanel.classList.add('opacity-0', 'translate-y-4');
            logContainer.innerHTML = '<div class="absolute bottom-0 w-full h-8 bg-gradient-to-t from-white to-transparent"></div>'; // Reset logs

            // 2. Tareas simuladas (Timeline)
            const tasks = [
                { time: 500,  percent: 15, text: "Conectando con API de Jira Software..." },
                { time: 1200, percent: 30, text: "Extrayendo tareas completadas del Sprint Activo." },
                { time: 2000, percent: 45, text: "Analizando Google Calendar en busca de reuniones." },
                { time: 2800, percent: 60, text: "Cruzando eventos de calendario con registros de Jira." },
                { time: 3600, percent: 75, text: "Generando borrador de reporte semanal (Google Docs)." },
                { time: 4400, percent: 90, text: "Actualizando hoja de seguimiento (Google Sheets)." },
                { time: 5000, percent: 100, text: "Sincronización finalizada con éxito." }
            ];

            // 3. Ejecutar simulación
            tasks.forEach(task => {
                setTimeout(() => {
                    // Actualizar barra y texto principal
                    progressBar.style.width = task.percent + '%';
                    progressPercent.innerText = task.percent + '%';
                    progressText.innerText = task.text;

                    // Añadir al log histórico
                    const li = document.createElement('li');
                    li.innerHTML = `<i class="fa-solid fa-angle-right text-brand-300 mr-1"></i> ${task.text}`;
                    logContainer.prepend(li); // Insertar arriba

                    // Si es la última tarea, mostrar resultados
                    if (task.percent === 100) {
                        setTimeout(() => {
                            progressContainer.classList.add('hidden');
                            resultsPanel.classList.remove('hidden');
                            // Delay for Tailwind animation
                            setTimeout(() => {
                                resultsPanel.classList.remove('opacity-0', 'translate-y-4');
                                resultsPanel.classList.add('opacity-100', 'translate-y-0');
                            }, 50);
                        }, 800);
                    }
                }, task.time);
            });
        }
